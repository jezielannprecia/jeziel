const nameInput = document.getElementById("name"); const statusInput = document.getElementById("status"); const deadlineInput = document.getElementById("deadline"); const searchInput = document.getElementById("search"); const listElement = document.getElementById("list"); let data = []; try { data = JSON.parse(localStorage.getItem("enrolleEase") || "[]"); if (!Array.isArray(data)) data = []; } catch (error) { data = []; } function save() { try { localStorage.setItem("enrolleEase", JSON.stringify(data)); render(); } catch (error) { alert("Unable to save data in this browser."); } } function add() { const n = nameInput.value.trim(); const s = statusInput.value; const d = deadlineInput.value; if (!n) { alert("Please enter a requirement name!"); nameInput.focus(); return; } data.unshift({ n: n, s: s, d: d }); nameInput.value = ""; deadlineInput.value = ""; save(); alert("Enrollment requirement added successfully!"); } function edit(i) { const n = prompt("Edit requirement:", data[i].n); if (n === null || !n.trim()) return; const s = prompt( "Status: Pending or Completed", data[i].s ); if (s === null) return; if (!["Pending", "Completed"].includes(s)) { alert("Please enter Pending or Completed."); return; } data[i].n = n.trim(); data[i].s = s; save(); } function del(i) { if (confirm("Are you sure you want to delete this requirement?")) { data.splice(i, 1); save(); } } function render() { const q = searchInput.value.toLowerCase(); listElement.innerHTML = data .map((x, i) => ({...x, i})) .filter(x => x.n.toLowerCase().includes(q)) .map(x => `

${escapeHTML(x.n)}

${escapeHTML(x.s)}

Deadline: ${escapeHTML(x.d || "Not specified")}

`).join("") || "No requirements added."; } function escapeHTML(value) { return String(value).replace(/[&<>"']/g, c => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[c]); } render();
